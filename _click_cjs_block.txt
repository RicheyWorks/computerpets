test("Click rights a window stool as a bark plate: walk onto the stool, sit, right once, sit the plate, then leave", () => {
  const WIN_B = { id: "hw2", x: 980, y: 90, width: 300, height: 360 };
  assert.equal(P.playFor("click_beetle"), "right");
  assert.equal(P.RIGHT, "right");
  assert.notEqual(P.playFor("click_beetle"), "click");
  assert.notEqual(P.playFor("click_beetle"), "snap");
  assert.notEqual(P.playFor("click_beetle"), "flip");
  assert.notEqual(P.playFor("click_beetle"), "glow");
  assert.notEqual(P.playFor("click_beetle"), "flash");
  assert.notEqual(P.playFor("click_beetle"), "spring");
  assert.notEqual(P.playFor("click_beetle"), "jump");
  assert.notEqual(P.playFor("click_beetle"), "drill");
  assert.notEqual(P.playFor("click_beetle"), "crack");
  assert.notEqual(P.playFor("click_beetle"), "tick");
  assert.notEqual(P.playFor("click_beetle"), "sill");
  assert.equal(P.playFor("acorn_weevil"), "drill");
  assert.equal(P.DRILL, "drill");
  assert.equal(P.playFor("relay_dragon"), "click");
  assert.equal(P.CLICK, "click");
  assert.equal(P.playFor("venus_flytrap"), "count");
  assert.equal(P.playFor("snapper"), "snap");
  assert.equal(P.playFor("hognose"), "flip");
  assert.equal(P.playFor("firefly"), "glow");
  assert.equal(P.playFor("robber_fly"), "sill");
  const target = P.pickTarget([WIN], 80, "click_beetle", WORK, P.SPRITE);
  assert.ok(target);
  assert.equal(target.kind, "right");
  assert.equal(target.side, "barkplate");
  assert.equal(target.leave, "righted");
  assert.notEqual(target.kind, "click");
  assert.notEqual(target.kind, "snap");
  assert.notEqual(target.kind, "flip");
  assert.notEqual(target.kind, "glow");
  assert.notEqual(target.kind, "drill");
  assert.notEqual(target.kind, "sill");
  assert.ok(target.holdLift >= 0, "she rights a window stool as a bark plate");
  assert.ok(P.DUR.rightHold > P.DUR.right * 1.2, "the hold is the sit; the right is the tell");
  assert.ok(P.DUR.rightOn > 1.0, "a walk onto the stool, not the right");
  assert.ok(P.DUR.rightOn !== P.DUR.drillOn);
  assert.ok(P.DUR.rightOn !== P.DUR.cerciOn);
  assert.ok(P.DUR.rightOn !== P.DUR.flipOn);
  assert.ok(P.DUR.rightOn !== P.DUR.jumpOn);
  assert.ok(P.DUR.rightOn !== P.DUR.snapOn);
  assert.ok(P.DUR.rightOn !== P.DUR.sillHop);
  assert.ok(P.DUR.right !== P.DUR.drill);
  assert.ok(P.DUR.right !== P.DUR.flip);
  assert.ok(P.DUR.right !== P.DUR.flipRoll);
  assert.ok(P.DUR.right !== P.DUR.jump);
  assert.ok(P.DUR.right !== P.DUR.snap);
  assert.ok(P.DUR.rightHold !== P.DUR.drillHold);
  assert.ok(P.DUR.rightHold !== P.DUR.flip);
  assert.ok(P.DUR.rightOff !== P.DUR.drillOff);
  assert.ok(P.DUR.rightOff !== P.DUR.flipOff);
  const plate = P.rightPoint(WIN, P.SPRITE, WORK);
  const cup = P.drillPoint(WIN, P.SPRITE, WORK);
  const bark = P.cerciPoint(WIN, P.SPRITE, WORK);
  const flip = P.flipPoint(WIN, P.SPRITE, WORK);
  const jump = P.jumpPoint(WIN, P.SPRITE, WORK);
  assert.ok(plate.lift >= 0, "the window stool as a bark plate, not the sky");
  assert.ok(Math.abs(plate.x - cup.x) > 12 || Math.abs(plate.lift - cup.lift) > 2, "not Snout's acorn-cup drill");
  assert.ok(Math.abs(plate.x - bark.x) > 12 || Math.abs(plate.lift - bark.lift) > 2, "not Forceps's bark-dish cerci");
  assert.ok(Math.abs(plate.x - flip.x) > 12 || Math.abs(plate.lift - flip.lift) > 2, "not Bluff's eraser-dish flip");
  assert.ok(Math.abs(plate.x - jump.x) > 12 || Math.abs(plate.lift - jump.lift) > 4, "not Vault's grass-plate jump");
  const tiny = P.pickTarget([{ id: "tiny", x: 200, y: 80, width: 120, height: 50 }], 80, "click_beetle", WORK, P.SPRITE);
  assert.equal(tiny, null, "a real window stool, not a thin strip");
  const short = P.pickTarget([{ id: "short", x: 200, y: 80, width: 168, height: 74 }], 80, "click_beetle", WORK, P.SPRITE);
  assert.equal(short, null, "a real bark plate, not Brood's emerge gate");
  const thin = P.pickTarget([{ id: "thin", x: 200, y: 80, width: 184, height: 150 }], 80, "click_beetle", WORK, P.SPRITE);
  assert.equal(thin, null, "a real bark-plate stool, not a thinner stool");
  const okRight = P.pickTarget([{ id: "right", x: 200, y: 80, width: 184, height: 152 }], 80, "click_beetle", WORK, P.SPRITE);
  assert.ok(okRight, "a real window stool as a bark plate");
  const drillOk = P.pickTarget([{ id: "drill", x: 200, y: 80, width: 186, height: 150 }], 80, "acorn_weevil", WORK, P.SPRITE);
  assert.ok(drillOk, "Snout still takes an acorn cup");
  const cerciOk = P.pickTarget([{ id: "cerci", x: 200, y: 80, width: 188, height: 154 }], 80, "earwig", WORK, P.SPRITE);
  assert.ok(cerciOk, "Forceps still takes a bark dish");
  const walkOn = P.rightOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  const drillOn = P.drillOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  const flipOn = P.flipOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  assert.ok(walkOn.lift >= 0, "she walks onto the stool as a bark plate");
  assert.ok(walkOn.rot !== drillOn.rot, "a walk onto the stool, not Snout drill");
  assert.ok(walkOn.rot !== flipOn.rot, "a walk onto the stool, not Bluff flip");
  const rightPose = P.rightPath(0.3);
  const drillPose = P.drillPath(0.3);
  const flipPose = P.flipRollPath(0.3);
  const jumpPose = P.jumpPath(0.3);
  assert.ok(rightPose.lift > 4 || Math.abs(rightPose.rot) > 40, "she rights with a click; right is the tell");
  assert.ok(rightPose.rot !== drillPose.rot, "right, not drill");
  assert.ok(rightPose.rot !== flipPose.rot, "right, not Bluff flip");
  assert.ok(rightPose.rot !== jumpPose.rot, "right, not Vault jump");
  const hold = P.rightHoldPath(0.5);
  assert.ok(Math.abs(hold.rot + 2.8) < 0.2, "she holds the sit on the plate");
  assert.ok(Math.abs(hold.x - 0.1) < 0.05, "she stays on the bark plate");
  assert.ok(hold.lift < 2, "sit on the stool, not a righting pop");
  const off0 = P.rightOffPath(0, { x: plate.x, lift: plate.lift + 0.6, rot: -2.8 }, { x: plate.x + 50, lift: 0 });
  const offMid = P.rightOffPath(0.5, { x: plate.x, lift: plate.lift + 0.6, rot: -2.8 }, { x: plate.x + 50, lift: 0 });
  const off1 = P.rightOffPath(1, { x: plate.x, lift: plate.lift + 0.6, rot: -2.8 }, { x: plate.x + 50, lift: 0 });
  assert.ok(Math.abs(off0.x - plate.x) < 2);
  assert.ok(Math.abs(offMid.x - plate.x) > 8, "a walk leave off the bark plate");
  assert.ok(Math.abs(off1.lift) < 3);
  let play = P.beginPlay(target, target.approachX);
  const seen = new Set();
  const windowIds = new Set();
  for (let i = 0; i < 2400 && play.phase !== "done"; i++) {
    seen.add(play.phase);
    if (play.target && play.target.id) windowIds.add(play.target.id);
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN, WIN_B], WORK, P.SPRITE, { cmd: "idle" });
    assert.notEqual(play.phase, "drill");
    assert.notEqual(play.phase, "flip");
    assert.notEqual(play.phase, "jump");
    assert.notEqual(play.phase, "snap");
    assert.notEqual(play.phase, "glow");
    assert.notEqual(play.phase, "sill-walk");
    if (play.phase === "right") {
      assert.ok(play.lift !== undefined, "she rights on the window stool");
    }
  }
  assert.ok(seen.has("right-on"));
  assert.ok(seen.has("right"));
  assert.ok(seen.has("right-hold"));
  assert.ok(seen.has("right-off"));
  assert.ok(seen.has("approach") || seen.has("walk") || seen.has("land"));
  assert.equal(windowIds.size, 1, "one window; bounds-only");
  assert.equal(play.phase, "done");
});

test("a moved window refits Click's bark-plate right; sleep, card, and hide abort; Click never starts asleep", () => {
  assert.equal(P.canStart({ asleep: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ card: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ hidden: true, cmd: "wander" }), false);
  assert.equal(P.shouldAbort({ cmd: "sleep" }), true);
  assert.equal(P.shouldAbort({ cmd: "hide" }), true);
  assert.equal(P.shouldAbort({ card: true, cmd: "idle" }), true);
  const target = P.pickTarget([WIN], 200, "click_beetle", WORK, P.SPRITE);
  assert.ok(target);
  let play = P.beginPlay(target, target.approachX);
  for (let i = 0; i < 900 && play.phase !== "right"; i++) {
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN], WORK, P.SPRITE, { cmd: "idle" });
  }
  assert.equal(play.phase, "right");
  const beforeX = play.target.holdX;
  const moved = { ...WIN, x: WIN.x + 140 };
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { cmd: "idle" });
  assert.equal(play.phase, "right");
  assert.ok(Math.abs(play.target.holdX - beforeX) > 40, "refit follows the moved window stool");
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { asleep: true, cmd: "sleep" });
  assert.ok(play.phase === "drop" || play.phase === "done" || play.phase === "right-off");
  assert.equal(play.abort, true);
});
