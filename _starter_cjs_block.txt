
test("Starter blooms a damp pane as a yeast film: walk onto the pane, sit the warm damp glass, bloom once, sit the soft culture film, then leave", () => {
  const WIN_B = { id: "hw2", x: 980, y: 90, width: 300, height: 360 };
  assert.equal(P.playFor("yeast"), "bloom");
  assert.equal(P.BLOOM, "bloom");
  assert.notEqual(P.playFor("yeast"), "drip");
  assert.notEqual(P.playFor("yeast"), "shelf");
  assert.notEqual(P.playFor("yeast"), "cloud");
  assert.notEqual(P.playFor("yeast"), "warts");
  assert.notEqual(P.playFor("yeast"), "teeth");
  assert.notEqual(P.playFor("yeast"), "zones");
  assert.notEqual(P.playFor("yeast"), "fork");
  assert.notEqual(P.playFor("yeast"), "hollow");
  assert.notEqual(P.playFor("yeast"), "dry");
  assert.notEqual(P.playFor("yeast"), "foam");
  assert.notEqual(P.playFor("yeast"), "culture");
  assert.notEqual(P.playFor("yeast"), "sill");
  assert.equal(P.playFor("chicken_of_woods"), "drip");
  assert.equal(P.DRIP, "drip");
  assert.equal(P.playFor("puffball"), "cloud");
  assert.equal(P.CLOUD, "cloud");
  assert.equal(P.playFor("oyster"), "shelf");
  assert.equal(P.SHELF, "shelf");
  assert.equal(P.playFor("lions_mane"), "teeth");
  assert.equal(P.playFor("turkey_tail"), "zones");
  assert.equal(P.playFor("chanterelle"), "fork");
  assert.equal(P.playFor("morel"), "hollow");
  assert.equal(P.playFor("fly_agaric"), "warts");
  assert.equal(P.playFor("lichen"), "sill");
  const target = P.pickTarget([WIN], 80, "yeast", WORK, P.SPRITE);
  assert.ok(target);
  assert.equal(target.kind, "bloom");
  assert.equal(target.side, "yeastfilm");
  assert.equal(target.leave, "bloomed");
  assert.notEqual(target.kind, "drip");
  assert.notEqual(target.kind, "dry");
  assert.notEqual(target.kind, "sill");
  assert.ok(target.holdLift >= 0, "she blooms a damp pane as a yeast film");
  assert.ok(P.DUR.bloomHold > P.DUR.bloom * 1.2, "the hold is the sit; bloom is the tell");
  assert.ok(P.DUR.bloomOn > 1.0, "a walk onto the pane, not a cling");
  assert.ok(P.DUR.bloomOn !== P.DUR.dripOn);
  assert.ok(P.DUR.bloomOn !== P.DUR.shelfOn);
  assert.ok(P.DUR.bloomOn !== P.DUR.cloudOn);
  assert.ok(P.DUR.bloomOn !== P.DUR.dryOn);
  assert.ok(P.DUR.bloomOn !== P.DUR.sillHop);
  assert.ok(P.DUR.bloom !== P.DUR.drip);
  assert.ok(P.DUR.bloom !== P.DUR.cloud);
  assert.ok(P.DUR.bloomHold !== P.DUR.dripHold);
  assert.ok(P.DUR.bloomOff !== P.DUR.dripOff);
  const film = P.bloomPoint(WIN, P.SPRITE, WORK);
  const wood = P.dripPoint(WIN, P.SPRITE, WORK);
  const timber = P.shelfPoint(WIN, P.SPRITE, WORK);
  const dish = P.cloudPoint(WIN, P.SPRITE, WORK);
  const cup = P.wartsPoint(WIN, P.SPRITE, WORK);
  const dry = P.dryPoint(WIN, P.SPRITE, WORK);
  assert.ok(film.lift >= 0, "the damp pane as a yeast film, not the sky");
  assert.ok(Math.abs(film.x - wood.x) > 12 || Math.abs(film.lift - wood.lift) > 1, "not Flame's right stile drip");
  assert.ok(Math.abs(film.x - timber.x) > 12 || Math.abs(film.lift - timber.lift) > 1, "not Frill's left timber shelf");
  assert.ok(Math.abs(film.x - dish.x) > 12 || Math.abs(film.lift - dish.lift) > 1, "not Puff's apron cloud");
  assert.ok(Math.abs(film.x - cup.x) > 12 || Math.abs(film.lift - cup.lift) > 1, "not Cap's moss-cup warts");
  assert.ok(Math.abs(film.x - dry.x) > 8 || Math.abs(film.lift - dry.lift) > 1, "not Tun's dry moss film");
  const tiny = P.pickTarget([{ id: "tiny", x: 200, y: 80, width: 120, height: 50 }], 80, "yeast", WORK, P.SPRITE);
  assert.equal(tiny, null, "a real damp pane, not a thin strip");
  const short = P.pickTarget([{ id: "short", x: 200, y: 80, width: 184, height: 154 }], 80, "yeast", WORK, P.SPRITE);
  assert.equal(short, null, "a real yeast film pane, not a thinner frame");
  const okBloom = P.pickTarget([{ id: "bloom", x: 200, y: 80, width: 186, height: 156 }], 80, "yeast", WORK, P.SPRITE);
  assert.ok(okBloom, "a real damp pane as a yeast film");
  const dripOk = P.pickTarget([{ id: "drip", x: 200, y: 80, width: 192, height: 210 }], 80, "chicken_of_woods", WORK, P.SPRITE);
  assert.ok(dripOk, "Flame still takes a warm wood stile");
  const walkOn = P.bloomOnPath(0.25, { x: 40, lift: 0 }, { x: film.x, lift: film.lift });
  const dripOn = P.dripOnPath(0.25, { x: 40, lift: 0 }, { x: film.x, lift: film.lift });
  assert.ok(walkOn.lift >= 0, "she walks onto the damp pane as a yeast film");
  assert.ok(walkOn.rot !== dripOn.rot, "a walk onto the pane, not Flame drip");
  const bloomPose = P.bloomPath(0.3);
  const dripPose = P.dripPath(0.3);
  const cloudPose = P.cloudPath(0.3);
  assert.ok(Math.abs(bloomPose.lift) > 0.3 || Math.abs(bloomPose.rot) > 1, "she blooms once; bloom is the tell");
  assert.ok(bloomPose.rot !== dripPose.rot, "bloom, not drip");
  assert.ok(bloomPose.rot !== cloudPose.rot, "bloom, not cloud");
  const hold = P.bloomHoldPath(0.5);
  assert.ok(Math.abs(hold.rot + 2.4) < 0.2, "she holds the sit on the yeast film");
  assert.ok(Math.abs(hold.x - 0.22) < 0.05, "she stays on the soft culture film");
  assert.ok(hold.lift < 1.0, "sit on the pane, not a bloom peak");
  const off0 = P.bloomOffPath(0, { x: film.x, lift: film.lift + 0.24, rot: -2.4 }, { x: film.x + 50, lift: 0 });
  const offMid = P.bloomOffPath(0.5, { x: film.x, lift: film.lift + 0.24, rot: -2.4 }, { x: film.x + 50, lift: 0 });
  const off1 = P.bloomOffPath(1, { x: film.x, lift: film.lift + 0.24, rot: -2.4 }, { x: film.x + 50, lift: 0 });
  assert.ok(Math.abs(off0.x - film.x) < 2);
  assert.ok(Math.abs(offMid.x - film.x) > 8, "a walk leave off the yeast film");
  assert.ok(Math.abs(off1.lift) < 3);
  let play = P.beginPlay(target, target.approachX);
  const seen = new Set();
  const windowIds = new Set();
  for (let i = 0; i < 2400 && play.phase !== "done"; i++) {
    seen.add(play.phase);
    if (play.target && play.target.id) windowIds.add(play.target.id);
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN, WIN_B], WORK, P.SPRITE, { cmd: "idle" });
    assert.notEqual(play.phase, "drip");
    assert.notEqual(play.phase, "shelf");
    assert.notEqual(play.phase, "cloud");
    assert.notEqual(play.phase, "sill-walk");
    if (play.phase === "bloom") {
      assert.ok(play.lift !== undefined, "she blooms on the damp pane");
    }
  }
  assert.ok(seen.has("bloom-on"));
  assert.ok(seen.has("bloom"));
  assert.ok(seen.has("bloom-hold"));
  assert.ok(seen.has("bloom-off"));
  assert.ok(seen.has("approach") || seen.has("walk") || seen.has("land"));
  assert.equal(windowIds.size, 1, "one window; bounds-only");
  assert.equal(play.phase, "done");
});

test("a moved window refits Starter's yeast-film bloom; sleep, card, and hide abort; Starter never starts asleep", () => {
  assert.equal(P.canStart({ asleep: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ card: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ hidden: true, cmd: "wander" }), false);
  assert.equal(P.shouldAbort({ cmd: "sleep" }), true);
  assert.equal(P.shouldAbort({ cmd: "hide" }), true);
  assert.equal(P.shouldAbort({ card: true, cmd: "idle" }), true);
  const target = P.pickTarget([WIN], 200, "yeast", WORK, P.SPRITE);
  assert.ok(target);
  let play = P.beginPlay(target, target.approachX);
  for (let i = 0; i < 900 && play.phase !== "bloom"; i++) {
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN], WORK, P.SPRITE, { cmd: "idle" });
  }
  assert.equal(play.phase, "bloom");
  const beforeX = play.target.holdX;
  const moved = { ...WIN, x: WIN.x + 140 };
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { cmd: "idle" });
  assert.equal(play.phase, "bloom");
  assert.ok(Math.abs(play.target.holdX - beforeX) > 40, "refit follows the moved yeast film");
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { asleep: true, cmd: "sleep" });
  assert.ok(play.phase === "drop" || play.phase === "done" || play.phase === "bloom-off");
  assert.equal(play.abort, true);
});
