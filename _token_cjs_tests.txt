test("Token flats a sill pan as a sand plate: walk onto the pan, sit the flat, then leave", () => {
  const WIN_B = { id: "hw2", x: 980, y: 90, width: 300, height: 360 };
  assert.equal(P.playFor("sand_dollar"), "flat");
  assert.equal(P.FLAT, "flat");
  assert.notEqual(P.playFor("sand_dollar"), "token");
  assert.notEqual(P.playFor("sand_dollar"), "bury");
  assert.notEqual(P.playFor("sand_dollar"), "sand");
  assert.notEqual(P.playFor("sand_dollar"), "circle");
  assert.notEqual(P.playFor("sand_dollar"), "open");
  assert.notEqual(P.playFor("sand_dollar"), "reef");
  assert.notEqual(P.playFor("sand_dollar"), "rock");
  assert.notEqual(P.playFor("sand_dollar"), "signal");
  assert.notEqual(P.playFor("sand_dollar"), "sill");
  assert.equal(P.playFor("squirrel"), "bury");
  assert.equal(P.BURY, "bury");
  assert.equal(P.playFor("ghost_crab"), "sand");
  assert.equal(P.SAND, "sand");
  assert.equal(P.playFor("periwinkle"), "rock");
  assert.equal(P.ROCK, "rock");
  assert.equal(P.playFor("chiton"), "eight");
  assert.equal(P.EIGHT, "eight");
  assert.equal(P.playFor("barnacle"), "cirri");
  assert.equal(P.CIRRI, "cirri");
  assert.equal(P.playFor("limpet"), "clamp");
  assert.equal(P.CLAMP, "clamp");
  assert.equal(P.playFor("fiddler_crab"), "signal");
  assert.equal(P.SIGNAL, "signal");
  assert.equal(P.playFor("goldfish"), "circle");
  assert.equal(P.CIRCLE, "circle");
  assert.equal(P.playFor("water_lily"), "open");
  assert.equal(P.OPEN, "open");
  assert.equal(P.playFor("sea_star"), "reef");
  assert.equal(P.REEF, "reef");
  assert.equal(P.playFor("sea_urchin"), "sill");
  assert.equal(P.DUR.rockOn, 2.80, "Spire rock durations stay");
  assert.equal(P.DUR.eightOn, 2.73, "Mail eight durations stay");
  assert.equal(P.DUR.cirriOn, 2.66, "Cement cirri durations stay");
  assert.equal(P.DUR.clampOn, 2.59, "Cone clamp durations stay");
  assert.equal(P.DUR.sandOn, 2.52, "Pale sand durations stay");
  assert.equal(P.DUR.signalOn, 2.45, "Wave signal durations stay");
  assert.equal(P.DUR.buryOn, 0.42, "Cache bury durations stay");
  const target = P.pickTarget([WIN], 80, "sand_dollar", WORK, P.SPRITE);
  assert.ok(target);
  assert.equal(target.kind, "flat");
  assert.equal(target.side, "sandplate");
  assert.equal(target.leave, "flats");
  assert.notEqual(target.kind, "token");
  assert.notEqual(target.kind, "bury");
  assert.notEqual(target.kind, "sand");
  assert.notEqual(target.kind, "circle");
  assert.notEqual(target.kind, "open");
  assert.notEqual(target.kind, "reef");
  assert.notEqual(target.kind, "signal");
  assert.notEqual(target.kind, "sill");
  assert.ok(target.holdLift > 28, "she flats a sill pan as a sand plate, not the floor");
  assert.ok(P.DUR.flatHold > P.DUR.flat, "the hold is the sit after; the flat is the tell");
  assert.ok(P.DUR.flatOn > 1.0, "a walk onto the pan, not the flat");
  assert.ok(P.DUR.flatOn !== P.DUR.rockOn);
  assert.ok(P.DUR.flatOn !== P.DUR.signalOn);
  assert.ok(P.DUR.flatOn !== P.DUR.sandOn);
  assert.ok(P.DUR.flatOn !== P.DUR.buryOn);
  assert.ok(P.DUR.flatOn !== P.DUR.paddleOn);
  assert.ok(P.DUR.flatOn !== P.DUR.rinseOn);
  assert.ok(P.DUR.flatOn !== P.DUR.showOn);
  assert.ok(P.DUR.flatOn !== P.DUR.fillOn);
  assert.ok(P.DUR.flatOn !== P.DUR.sillHop);
  assert.ok(P.DUR.flat !== P.DUR.rock);
  assert.ok(P.DUR.flat !== P.DUR.signal);
  assert.ok(P.DUR.flat !== P.DUR.sand);
  assert.ok(P.DUR.flat !== P.DUR.bury);
  assert.ok(P.DUR.flatHold !== P.DUR.rockHold);
  assert.ok(P.DUR.flatHold !== P.DUR.signalHold);
  assert.ok(P.DUR.flatOff !== P.DUR.rockOff);
  assert.ok(P.DUR.flatOff !== P.DUR.sillDown);
  const plate = P.flatPoint(WIN, P.SPRITE, WORK);
  const marsh = P.signalPoint(WIN, P.SPRITE, WORK);
  const current = P.paddlePoint(WIN, P.SPRITE, WORK);
  const bowl = P.rinsePoint(WIN, P.SPRITE, WORK);
  const brackish = P.showPoint(WIN, P.SPRITE, WORK);
  const bog = P.fillPoint(WIN, P.SPRITE, WORK);
  const dry = P.sandPoint(WIN, P.SPRITE, WORK);
  const oak = P.buryPoint(WIN, P.SPRITE, WORK);
  assert.ok(plate.lift > 28, "the sill pan as a sand plate, not the floor");
  assert.ok(Math.abs(plate.lift - marsh.lift) < 2, "the same sill pan Wave signals; the pose is a flat");
  assert.ok(Math.abs(plate.x - marsh.x) > 8, "not Wave's sill-pan signal");
  assert.ok(Math.abs(plate.lift - current.lift) < 2, "the same sill pan Spoon paddles; the pose is a flat");
  assert.ok(Math.abs(plate.x - current.x) > 8, "not Spoon's sill-pan paddle");
  assert.ok(Math.abs(plate.lift - bowl.lift) < 2, "the same sill pan Wash rinses; the pose is a flat");
  assert.ok(Math.abs(plate.x - bowl.x) > 8, "not Wash's sill-pan rinse");
  assert.ok(Math.abs(plate.lift - brackish.lift) < 2, "the same sill pan Jaw shows; the pose is a flat");
  assert.ok(Math.abs(plate.x - brackish.x) > 8, "not Jaw's sill-pan show");
  assert.ok(Math.abs(plate.lift - bog.lift) < 2, "the same sill pan Well fills; the pose is a flat");
  assert.ok(Math.abs(plate.x - bog.x) > 8, "not Well's sill-pan fill");
  assert.ok(Math.abs(plate.x - dry.x) > 8 || Math.abs(plate.lift - dry.lift) > 8, "not Pale's window-stool sand");
  assert.ok(Math.abs(plate.x - oak.x) > 8 || Math.abs(plate.lift - oak.lift) > 8, "not Cache's window-stool bury");
  const tiny = P.pickTarget([{ id: "tiny", x: 200, y: 80, width: 120, height: 50 }], 80, "sand_dollar", WORK, P.SPRITE);
  assert.equal(tiny, null, "a real sill pan, not a thin strip");
  const short = P.pickTarget([{ id: "short", x: 200, y: 80, width: 190, height: 80 }], 80, "sand_dollar", WORK, P.SPRITE);
  assert.equal(short, null, "a real sill pan, not a short sash");
  const thin = P.pickTarget([{ id: "thin", x: 200, y: 80, width: 190, height: 182 }], 80, "sand_dollar", WORK, P.SPRITE);
  assert.equal(thin, null, "Token needs a real sill pan, not a shallower mouth");
  const pan = P.pickTarget([{ id: "pan", x: 200, y: 80, width: 190, height: 184 }], 80, "sand_dollar", WORK, P.SPRITE);
  assert.ok(pan, "a real sill pan as a sand plate");
  const waveOk = P.pickTarget([{ id: "pan", x: 200, y: 80, width: 190, height: 184 }], 80, "fiddler_crab", WORK, P.SPRITE);
  assert.ok(waveOk, "Wave still takes the pan");
  const spoonOk = P.pickTarget([{ id: "pan", x: 200, y: 80, width: 190, height: 184 }], 80, "paddlefish", WORK, P.SPRITE);
  assert.ok(spoonOk, "Spoon still takes the pan");
  const washOk = P.pickTarget([{ id: "pan", x: 200, y: 80, width: 190, height: 184 }], 80, "raccoon", WORK, P.SPRITE);
  assert.ok(washOk, "Wash still takes the pan");
  const jawOk = P.pickTarget([{ id: "pan", x: 200, y: 80, width: 190, height: 184 }], 80, "crocodile", WORK, P.SPRITE);
  assert.ok(jawOk, "Jaw still takes the pan");
  const wellOk = P.pickTarget([{ id: "pan", x: 200, y: 80, width: 190, height: 184 }], 80, "pitcher", WORK, P.SPRITE);
  assert.ok(wellOk, "Well still takes the pan");
  const walkOn = P.flatOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  const signalOn = P.signalOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  const paddleOn = P.paddleOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  const rinseOn = P.rinseOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  const showOn = P.showOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  const sandOn = P.sandOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  assert.ok(walkOn.lift > 0, "she walks onto the pan as a sand plate");
  assert.ok(walkOn.rot !== signalOn.rot, "a walk onto the pan, not Wave signal");
  assert.ok(walkOn.rot !== paddleOn.rot, "a walk onto the pan, not Spoon paddle");
  assert.ok(walkOn.rot !== rinseOn.rot, "a walk onto the pan, not Wash rinse");
  assert.ok(walkOn.rot !== showOn.rot, "a walk onto the pan, not Jaw show");
  assert.ok(walkOn.rot !== sandOn.rot, "a walk onto the pan, not Pale sand");
  const pulse = P.flatPath(0.5);
  const signalPose = P.signalPath(0.5);
  const paddlePose = P.paddlePath(0.5);
  const rinsePose = P.rinsePath(0.5);

    }
  }

  assert.equal(windowIds.size, 1, "one window");
  assert.equal(play.phase, "done");
  assert.equal(play.lift, 0);
});
test("a moved window refits Token's sand-plate flat; sleep, card, and hide abort; Token never starts asleep", () => {
  assert.equal(P.canStart({ asleep: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ card: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ hidden: true, cmd: "wander" }), false);
  assert.equal(P.shouldAbort({ cmd: "sleep" }), true);
  assert.equal(P.shouldAbort({ cmd: "hide" }), true);
  assert.equal(P.shouldAbort({ card: true, cmd: "idle" }), true);
  const target = P.pickTarget([WIN], 200, "sand_dollar", WORK, P.SPRITE);
  let play = P.beginPlay(target, target.approachX);
  for (let i = 0; i < 900 && play.phase !== "flat"; i++) {
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN], WORK, P.SPRITE, { cmd: "idle" });
  }
  assert.equal(play.phase, "flat");
  const beforeX = play.target.holdX;
  const moved = { ...WIN, x: WIN.x + 140 };
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { cmd: "idle" });
  assert.equal(play.phase, "flat");
  assert.ok(Math.abs(play.target.holdX - beforeX) > 40);
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { asleep: true, cmd: "sleep" });
  assert.ok(play.phase === "drop" || play.phase === "done" || play.phase === "flat-off");
  assert.equal(play.abort, true);
});
