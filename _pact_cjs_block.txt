test("Pact plaques a cool stile as bark stone: walk onto the stile, sit the cool bark wood, plaque once, sit the two-kingdom share, then leave", () => {
  const WIN_B = { id: "hw2", x: 980, y: 90, width: 300, height: 360 };
  assert.equal(P.playFor("lichen"), "plaque");
  assert.equal(P.PLAQUE, "plaque");
  assert.notEqual(P.playFor("lichen"), "bloom");
  assert.notEqual(P.playFor("lichen"), "drip");
  assert.notEqual(P.playFor("lichen"), "shelf");
  assert.notEqual(P.playFor("lichen"), "cloud");
  assert.notEqual(P.playFor("lichen"), "warts");
  assert.notEqual(P.playFor("lichen"), "teeth");
  assert.notEqual(P.playFor("lichen"), "zones");
  assert.notEqual(P.playFor("lichen"), "fork");
  assert.notEqual(P.playFor("lichen"), "hollow");
  assert.notEqual(P.playFor("lichen"), "dry");
  assert.notEqual(P.playFor("lichen"), "crust");
  assert.notEqual(P.playFor("lichen"), "stain");
  assert.notEqual(P.playFor("lichen"), "paint");
  assert.notEqual(P.playFor("lichen"), "sill");
  assert.equal(P.playFor("yeast"), "bloom");
  assert.equal(P.BLOOM, "bloom");
  assert.equal(P.playFor("chicken_of_woods"), "drip");
  assert.equal(P.DRIP, "drip");
  assert.equal(P.playFor("puffball"), "cloud");
  assert.equal(P.CLOUD, "cloud");
  assert.equal(P.playFor("oyster"), "shelf");
  assert.equal(P.SHELF, "shelf");
  assert.equal(P.playFor("lions_mane"), "teeth");
  assert.equal(P.playFor("turkey_tail"), "zones");
  assert.equal(P.playFor("chanterelle"), "fork");
  assert.equal(P.playFor("morel"), "hollow");
  assert.equal(P.playFor("fly_agaric"), "warts");
  assert.equal(P.playFor("photovore"), "sill");
  const target = P.pickTarget([WIN], 80, "lichen", WORK, P.SPRITE);
  assert.ok(target);
  assert.equal(target.kind, "plaque");
  assert.equal(target.side, "barkstone");
  assert.equal(target.leave, "plaqued");
  assert.notEqual(target.kind, "bloom");
  assert.notEqual(target.kind, "drip");
  assert.notEqual(target.kind, "sill");
  assert.ok(target.holdLift >= 0, "she plaques a cool stile as bark stone");
  assert.ok(P.DUR.plaqueHold > P.DUR.plaque * 1.2, "the hold is the sit; plaque is the tell");
  assert.ok(P.DUR.plaqueOn > 1.0, "a walk onto the stile, not a cling");
  assert.ok(P.DUR.plaqueOn !== P.DUR.bloomOn);
  assert.ok(P.DUR.plaqueOn !== P.DUR.dripOn);
  assert.ok(P.DUR.plaqueOn !== P.DUR.shelfOn);
  assert.ok(P.DUR.plaqueOn !== P.DUR.cloudOn);
  assert.ok(P.DUR.plaqueOn !== P.DUR.sillHop);
  assert.ok(P.DUR.plaque !== P.DUR.bloom);
  assert.ok(P.DUR.plaque !== P.DUR.drip);
  assert.ok(P.DUR.plaqueHold !== P.DUR.bloomHold);
  assert.ok(P.DUR.plaqueOff !== P.DUR.bloomOff);
  const stone = P.plaquePoint(WIN, P.SPRITE, WORK);
  const film = P.bloomPoint(WIN, P.SPRITE, WORK);
  const wood = P.dripPoint(WIN, P.SPRITE, WORK);
  const timber = P.shelfPoint(WIN, P.SPRITE, WORK);
  const dish = P.cloudPoint(WIN, P.SPRITE, WORK);
  const cup = P.wartsPoint(WIN, P.SPRITE, WORK);
  const dry = P.dryPoint(WIN, P.SPRITE, WORK);
  assert.ok(stone.lift >= 0, "the cool stile as bark stone, not the sky");
  assert.ok(Math.abs(stone.x - film.x) > 12 || Math.abs(stone.lift - film.lift) > 1, "not Starter's damp yeast film");
  assert.ok(Math.abs(stone.x - wood.x) > 12 || Math.abs(stone.lift - wood.lift) > 1, "not Flame's right stile drip");
  assert.ok(Math.abs(stone.x - timber.x) > 8 || Math.abs(stone.lift - timber.lift) > 1, "not Frill's mid left timber shelf");
  assert.ok(Math.abs(stone.x - dish.x) > 12 || Math.abs(stone.lift - dish.lift) > 1, "not Puff's apron cloud");
  assert.ok(Math.abs(stone.x - cup.x) > 12 || Math.abs(stone.lift - cup.lift) > 1, "not Cap's moss-cup warts");
  assert.ok(Math.abs(stone.x - dry.x) > 8 || Math.abs(stone.lift - dry.lift) > 1, "not Tun's dry moss film");
  const tiny = P.pickTarget([{ id: "tiny", x: 200, y: 80, width: 120, height: 50 }], 80, "lichen", WORK, P.SPRITE);
  assert.equal(tiny, null, "a real cool stile, not a thin strip");
  const short = P.pickTarget([{ id: "short", x: 200, y: 80, width: 186, height: 166 }], 80, "lichen", WORK, P.SPRITE);
  assert.equal(short, null, "a real bark-stone stile, not a thinner frame");
  const okPlaque = P.pickTarget([{ id: "plaque", x: 200, y: 80, width: 188, height: 168 }], 80, "lichen", WORK, P.SPRITE);
  assert.ok(okPlaque, "a real cool stile as bark stone");
  const bloomOk = P.pickTarget([{ id: "bloom", x: 200, y: 80, width: 186, height: 156 }], 80, "yeast", WORK, P.SPRITE);
  assert.ok(bloomOk, "Starter still takes a yeast film pane");
  const walkOn = P.plaqueOnPath(0.25, { x: 40, lift: 0 }, { x: stone.x, lift: stone.lift });
  const bloomOn = P.bloomOnPath(0.25, { x: 40, lift: 0 }, { x: stone.x, lift: stone.lift });
  assert.ok(walkOn.lift >= 0, "she walks onto the cool stile as bark stone");
  assert.ok(walkOn.rot !== bloomOn.rot, "a walk onto the stile, not Starter bloom");
  const plaquePose = P.plaquePath(0.3);
  const bloomPose = P.bloomPath(0.3);
  const dripPose = P.dripPath(0.3);
  assert.ok(Math.abs(plaquePose.lift) > 0.2 || Math.abs(plaquePose.rot) > 0.8, "she plaques once; plaque is the tell");
  assert.ok(plaquePose.rot !== bloomPose.rot, "plaque, not bloom");
  assert.ok(plaquePose.rot !== dripPose.rot, "plaque, not drip");
  const hold = P.plaqueHoldPath(0.5);
  assert.ok(Math.abs(hold.rot + 1.7) < 0.2, "she holds the sit on the bark stone");
  assert.ok(Math.abs(hold.x - 0.14) < 0.05, "she stays on the two-kingdom share");
  assert.ok(hold.lift < 1.0, "sit on the stile, not a plaque peak");
  const off0 = P.plaqueOffPath(0, { x: stone.x, lift: stone.lift + 0.18, rot: -1.7 }, { x: stone.x + 50, lift: 0 });
  const offMid = P.plaqueOffPath(0.5, { x: stone.x, lift: stone.lift + 0.18, rot: -1.7 }, { x: stone.x + 50, lift: 0 });
  const off1 = P.plaqueOffPath(1, { x: stone.x, lift: stone.lift + 0.18, rot: -1.7 }, { x: stone.x + 50, lift: 0 });
  assert.ok(Math.abs(off0.x - stone.x) < 2);
  assert.ok(Math.abs(offMid.x - stone.x) > 8, "a walk leave off the bark stone");
  assert.ok(Math.abs(off1.lift) < 3);
  let play = P.beginPlay(target, target.approachX);
  const seen = new Set();
  const windowIds = new Set();
  for (let i = 0; i < 2400 && play.phase !== "done"; i++) {
    seen.add(play.phase);
    if (play.target && play.target.id) windowIds.add(play.target.id);
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN, WIN_B], WORK, P.SPRITE, { cmd: "idle" });
    assert.notEqual(play.phase, "bloom");
    assert.notEqual(play.phase, "drip");
    assert.notEqual(play.phase, "shelf");
    assert.notEqual(play.phase, "sill-walk");
    if (play.phase === "plaque") {
      assert.ok(play.lift !== undefined, "she plaques on the cool stile");
    }
  }
  assert.ok(seen.has("plaque-on"));
  assert.ok(seen.has("plaque"));
  assert.ok(seen.has("plaque-hold"));
  assert.ok(seen.has("plaque-off"));
  assert.ok(seen.has("approach") || seen.has("walk") || seen.has("land"));
  assert.equal(windowIds.size, 1, "one window; bounds-only");
  assert.equal(play.phase, "done");
});

test("a moved window refits Pact's bark-stone plaque; sleep, card, and hide abort; Pact never starts asleep", () => {
  assert.equal(P.canStart({ asleep: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ card: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ hidden: true, cmd: "wander" }), false);
  assert.equal(P.shouldAbort({ cmd: "sleep" }), true);
  assert.equal(P.shouldAbort({ cmd: "hide" }), true);
  assert.equal(P.shouldAbort({ card: true, cmd: "idle" }), true);
  const target = P.pickTarget([WIN], 200, "lichen", WORK, P.SPRITE);
  assert.ok(target);
  let play = P.beginPlay(target, target.approachX);
  for (let i = 0; i < 900 && play.phase !== "plaque"; i++) {
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN], WORK, P.SPRITE, { cmd: "idle" });
  }
  assert.equal(play.phase, "plaque");
  const beforeX = play.target.holdX;
  const moved = { ...WIN, x: WIN.x + 140 };
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { cmd: "idle" });
  assert.equal(play.phase, "plaque");
  assert.ok(Math.abs(play.target.holdX - beforeX) > 40, "refit follows the moved bark stone");
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { asleep: true, cmd: "sleep" });
  assert.ok(play.phase === "drop" || play.phase === "done" || play.phase === "plaque-off");
  assert.equal(play.abort, true);
});
