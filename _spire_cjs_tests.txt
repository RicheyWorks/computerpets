test("Spire rocks a sash jamb as a rock face: walk onto the jamb, graze the rock, then leave", () => {
  const WIN_B = { id: "hw2", x: 980, y: 90, width: 300, height: 360 };
  assert.equal(P.playFor("periwinkle"), "rock");
  assert.equal(P.ROCK, "rock");
  assert.notEqual(P.playFor("periwinkle"), "spire");
  assert.notEqual(P.playFor("periwinkle"), "rasp");
  assert.notEqual(P.playFor("periwinkle"), "eight");
  assert.notEqual(P.playFor("periwinkle"), "cirri");
  assert.notEqual(P.playFor("periwinkle"), "clamp");
  assert.notEqual(P.playFor("periwinkle"), "sill");
  assert.equal(P.playFor("pond_snail"), "rasp");
  assert.equal(P.RASP, "rasp");
  assert.equal(P.playFor("chiton"), "eight");
  assert.equal(P.EIGHT, "eight");
  assert.equal(P.playFor("barnacle"), "cirri");
  assert.equal(P.CIRRI, "cirri");
  assert.equal(P.playFor("limpet"), "clamp");
  assert.equal(P.CLAMP, "clamp");
  assert.equal(P.playFor("ghost_crab"), "sand");
  assert.equal(P.SAND, "sand");
  assert.equal(P.playFor("fiddler_crab"), "signal");
  assert.equal(P.SIGNAL, "signal");
  assert.equal(P.playFor("box_turtle"), "shut");
  assert.equal(P.SHUT, "shut");
  assert.equal(P.playFor("rosy_boa"), "stone");
  assert.equal(P.STONE, "stone");
  assert.equal(P.playFor("nautilus"), "rise");
  assert.equal(P.RISE, "rise");
  assert.equal(P.playFor("sand_dollar"), "sill");
  assert.equal(P.DUR.eightOn, 2.73, "Mail eight durations stay");
  assert.equal(P.DUR.cirriOn, 2.66, "Cement cirri durations stay");
  assert.equal(P.DUR.clampOn, 2.59, "Cone clamp durations stay");
  assert.equal(P.DUR.sandOn, 2.52, "Pale sand durations stay");
  assert.equal(P.DUR.signalOn, 2.45, "Wave signal durations stay");
  assert.equal(P.DUR.shutOn, 1.47, "Lid shut durations stay");
  assert.equal(P.DUR.raspOn, 2.69, "Whorl rasp durations stay");
  assert.equal(P.DUR.riseOn, P.DUR.riseOn, "Chamber rise durations stay");
  const target = P.pickTarget([WIN], 80, "periwinkle", WORK, P.SPRITE);
  assert.ok(target);
  assert.equal(target.kind, "rock");
  assert.equal(target.side, "rockface");
  assert.equal(target.leave, "grazed");
  assert.notEqual(target.kind, "spire");
  assert.notEqual(target.kind, "rasp");
  assert.notEqual(target.kind, "eight");
  assert.notEqual(target.kind, "cirri");
  assert.notEqual(target.kind, "clamp");
  assert.notEqual(target.kind, "sill");
  assert.ok(target.holdLift > 28, "she rocks a sash jamb as a rock face, not the floor");
  assert.ok(P.DUR.rockHold > P.DUR.rock, "the hold is the sit after; the rock is the graze");
  assert.ok(P.DUR.rockOn > 1.0, "a walk onto the jamb, not the graze");
  assert.ok(P.DUR.rockOn !== P.DUR.eightOn);
  assert.ok(P.DUR.rockOn !== P.DUR.cirriOn);
  assert.ok(P.DUR.rockOn !== P.DUR.clampOn);
  assert.ok(P.DUR.rockOn !== P.DUR.raspOn);
  assert.ok(P.DUR.rockOn !== P.DUR.bristleOn);
  assert.ok(P.DUR.rockOn !== P.DUR.skinOn);
  assert.ok(P.DUR.rockOn !== P.DUR.mountOn);
  assert.ok(P.DUR.rockOn !== P.DUR.riseOn);
  assert.ok(P.DUR.rockOn !== P.DUR.sillHop);
  assert.ok(P.DUR.rock !== P.DUR.eight);
  assert.ok(P.DUR.rock !== P.DUR.rasp);
  assert.ok(P.DUR.rock !== P.DUR.cirri);
  assert.ok(P.DUR.rock !== P.DUR.clamp);
  assert.ok(P.DUR.rockHold !== P.DUR.eightHold);
  assert.ok(P.DUR.rockHold !== P.DUR.raspHold);
  assert.ok(P.DUR.rockOff !== P.DUR.eightOff);
  assert.ok(P.DUR.rockOff !== P.DUR.sillDown);
  const jamb = P.rockPoint(WIN, P.SPRITE, WORK);
  const cling = P.clingPoint(WIN, P.SPRITE, WORK);
  const bristle = P.bristlePoint(WIN, P.SPRITE, WORK);
  const mount = P.mountPoint(WIN, P.SPRITE, WORK);
  const rise = P.risePoint(WIN, "left", 0, P.SPRITE, WORK);
  const rail = P.eightPoint(WIN, P.SPRITE, WORK);
  const rasp = P.raspPoint(WIN, P.SPRITE, WORK);
  const rim = P.clampPoint(WIN, P.SPRITE, WORK);
  const stile = P.cirriPoint(WIN, P.SPRITE, WORK);
  assert.ok(jamb.lift > 28, "the sash jamb as a rock face, not the floor");
  assert.ok(Math.abs(jamb.x - cling.x) > 2 || Math.abs(jamb.lift - cling.lift) > 8, "same jamb family as Sail; she rocks, she does not cling");
  assert.ok(Math.abs(jamb.x - bristle.x) > 2 || Math.abs(jamb.lift - bristle.lift) > 8, "same jamb family as Spine; she rocks, she does not bristle");
  assert.ok(Math.abs(jamb.x - mount.x) > 2 || Math.abs(jamb.lift - mount.lift) > 8, "same jamb family as Moth; she rocks, she does not mount");
  assert.ok(Math.abs(jamb.x - rise.x) > 2 || Math.abs(jamb.lift - rise.lift) > 8, "same jamb family as Chamber; she rocks, she does not rise");
  assert.ok(Math.abs(jamb.x - rail.x) > 8, "not Mail's meeting-rail eight");
  assert.ok(Math.abs(jamb.x - rasp.x) > 8 || Math.abs(jamb.lift - rasp.lift) > 8, "not Whorl's glass-rim rasp");
  assert.ok(Math.abs(jamb.x - rim.x) > 8 || Math.abs(jamb.lift - rim.lift) > 8, "not Cone's glass-rim clamp");
  assert.ok(Math.abs(jamb.x - stile.x) > 2 || Math.abs(jamb.lift - stile.lift) > 8, "not Cement's sash-stile cirri");
  const tiny = P.pickTarget([{ id: "tiny", x: 200, y: 80, width: 120, height: 50 }], 80, "periwinkle", WORK, P.SPRITE);
  assert.equal(tiny, null, "a real sash jamb, not a thin strip");
  const short = P.pickTarget([{ id: "short", x: 200, y: 80, width: 182, height: 80 }], 80, "periwinkle", WORK, P.SPRITE);
  assert.equal(short, null, "a real sash jamb, not a short sash");
  const thin = P.pickTarget([{ id: "thin", x: 200, y: 80, width: 182, height: 215 }], 80, "periwinkle", WORK, P.SPRITE);
  assert.equal(thin, null, "Spire needs a real sash jamb, not a shallower mouth");
  const face = P.pickTarget([{ id: "jamb", x: 200, y: 80, width: 182, height: 216 }], 80, "periwinkle", WORK, P.SPRITE);
  assert.ok(face, "a real sash jamb as a rock face");
  const sailOk = P.pickTarget([{ id: "jamb", x: 200, y: 80, width: 182, height: 216 }], 80, "colugo", WORK, P.SPRITE);
  assert.ok(sailOk, "Sail still takes the jamb");
  const spineOk = P.pickTarget([{ id: "jamb", x: 200, y: 80, width: 182, height: 216 }], 80, "porcupine", WORK, P.SPRITE);
  assert.ok(spineOk, "Spine still takes the jamb");
  const mothOk = P.pickTarget([{ id: "jamb", x: 200, y: 80, width: 182, height: 216 }], 80, "orchid", WORK, P.SPRITE);
  assert.ok(mothOk, "Moth still takes the jamb");
  const chamberOk = P.pickTarget([{ id: "jamb", x: 200, y: 80, width: 168, height: 228 }], 80, "nautilus", WORK, P.SPRITE);
  assert.ok(chamberOk, "Chamber still takes the jamb");
  const mailOk = P.pickTarget([{ id: "rail", x: 200, y: 80, width: 195, height: 193 }], 80, "chiton", WORK, P.SPRITE);
  assert.ok(mailOk, "Mail still takes the meeting rail");
  const cementOk = P.pickTarget([{ id: "stile", x: 200, y: 80, width: 191, height: 216 }], 80, "barnacle", WORK, P.SPRITE);
  assert.ok(cementOk, "Cement still takes the stile");
  const coneOk = P.pickTarget([{ id: "rim", x: 200, y: 80, width: 184, height: 178 }], 80, "limpet", WORK, P.SPRITE);
  assert.ok(coneOk, "Cone still takes the glass rim");
  const walkOn = P.rockOnPath(0.25, { x: 40, lift: 0 }, { x: jamb.x, lift: jamb.lift });
  const eightOn = P.eightOnPath(0.25, { x: 40, lift: 0 }, { x: jamb.x, lift: jamb.lift });
  const cirriOn = P.cirriOnPath(0.25, { x: 40, lift: 0 }, { x: jamb.x, lift: jamb.lift });
  const clampOn = P.clampOnPath(0.25, { x: 40, lift: 0 }, { x: jamb.x, lift: jamb.lift });
  const raspOn = P.raspOnPath(0.25, { x: 40, lift: 0 }, { x: jamb.x, lift: jamb.lift });
  const bristleOn = P.bristleOnPath(0.25, { x: 40, lift: 0 }, { x: jamb.x, lift: jamb.lift });
  assert.ok(walkOn.lift > 0, "she walks onto the jamb as a rock face");
  assert.ok(walkOn.rot !== eightOn.rot, "a walk onto the jamb, not Mail eight");
  assert.ok(walkOn.rot !== cirriOn.rot, "a walk onto the jamb, not Cement cirri");
  assert.ok(walkOn.rot !== clampOn.rot, "a walk onto the jamb, not Cone clamp");
  assert.ok(walkOn.rot !== raspOn.rot, "a walk onto the jamb, not Whorl rasp");
  assert.ok(walkOn.rot !== bristleOn.rot, "a walk onto the jamb, not Spine bristle");
  const pulse = P.rockPath(0.5);
  const eightPose = P.eightPath(0.5);
  const cirriPose = P.cirriPath(0.5);
  const clampPose = P.clampPath(0.5);
  const raspPose = P.raspPath(0.5);
  const bristlePose = P.bristlePath(0.5);
  const clingPose = P.clingPath(0.5);
  const mountPose = P.mountPath(0.5);
  assert.ok(pulse.lift > 0, "she grazes the rock; the rock is the tell");
  assert.ok(pulse.rot > 10, "a graze on the rock face");
  assert.ok(pulse.rot !== eightPose.rot, "a rock, not an eight");
  assert.ok(pulse.rot !== cirriPose.rot, "a rock, not a cirri");
  assert.ok(pulse.rot !== clampPose.rot, "a rock, not a clamp");
  assert.ok(pulse.rot !== raspPose.rot, "a rock, not a rasp");
  assert.ok(pulse.rot !== bristlePose.rot, "a rock, not a bristle");
  assert.ok(pulse.rot !== clingPose.rot, "a rock, not a cling");
  assert.ok(pulse.rot !== mountPose.rot, "a rock, not a mount");
  const hold = P.rockHoldPath(0.5);
  assert.ok(Math.abs(hold.rot - 12.1) < 0.2, "she holds after the graze, rock still the tell");
  assert.ok(Math.abs(hold.x - 2.5) < 0.2, "she stays on the rock face");
  assert.ok(hold.lift > 0, "a sit-hold on the jamb after the graze");
  const off0 = P.rockOffPath(0, { x: jamb.x, lift: jamb.lift, rot: 12.1 }, { x: jamb.x + 50, lift: 0 });
  const offMid = P.rockOffPath(0.5, { x: jamb.x, lift: jamb.lift, rot: 12.1 }, { x: jamb.x + 50, lift: 0 });
  const off1 = P.rockOffPath(1, { x: jamb.x, lift: jamb.lift, rot: 12.1 }, { x: jamb.x + 50, lift: 0 });
  assert.ok(Math.abs(off0.x - jamb.x) < 2);
  assert.ok(Math.abs(offMid.x - jamb.x) > 8, "a walk leave off the rock face");
  assert.ok(Math.abs(off1.lift) < 3);
  let play = P.beginPlay(target, target.approachX);
  const seen = new Set();
  const windowIds = new Set();
  for (let i = 0; i < 2400 && play.phase !== "done"; i++) {
    seen.add(play.phase);
    if (play.target && play.target.id) windowIds.add(play.target.id);
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN, WIN_B], WORK, P.SPRITE, { cmd: "idle" });
    assert.notEqual(play.phase, "eight");
    assert.notEqual(play.phase, "cirri");
    assert.notEqual(play.phase, "clamp");
    assert.notEqual(play.phase, "rasp");
    assert.notEqual(play.phase, "bristle");
    assert.notEqual(play.phase, "sill-walk");
    if (play.phase === "rock") {
      assert.equal(play.anim, "play");
    }
    if (play.phase === "rock-on") {
      assert.equal(play.anim, "walk");
    }
    if (play.phase === "rock-hold" && play.t > 0) {
      assert.equal(play.anim, "sit");
      assert.ok(Math.abs(play.lift - (play.target.holdLift + 1.9)) < 3, "she holds the graze after the rock on the jamb");
    }
    if (play.phase === "rock-off") {
      assert.equal(play.anim, "walk");
    }
  }
  assert.ok(seen.has("rock-on"));
  assert.ok(seen.has("rock"));
  assert.ok(seen.has("rock-hold"));
  assert.ok(seen.has("rock-off"));
  assert.ok(!seen.has("eight"), "Spire never uses Mail eight");
  assert.ok(!seen.has("cirri"), "Spire never uses Cement cirri");
  assert.ok(!seen.has("clamp"), "Spire never uses Cone clamp");
  assert.ok(!seen.has("rasp"), "Spire never uses Whorl rasp");
  assert.ok(!seen.has("bristle"), "Spire never uses Spine bristle");
  assert.equal(windowIds.size, 1, "one window");
  assert.equal(play.phase, "done");
  assert.equal(play.lift, 0);
});
test("a moved window refits Spire's rock-face graze; sleep, card, and hide abort; Spire never starts asleep", () => {
  assert.equal(P.canStart({ asleep: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ card: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ hidden: true, cmd: "wander" }), false);
  assert.equal(P.shouldAbort({ cmd: "sleep" }), true);
  assert.equal(P.shouldAbort({ cmd: "hide" }), true);
  assert.equal(P.shouldAbort({ card: true, cmd: "idle" }), true);
  const target = P.pickTarget([WIN], 200, "periwinkle", WORK, P.SPRITE);
  let play = P.beginPlay(target, target.approachX);
  for (let i = 0; i < 900 && play.phase !== "rock"; i++) {
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN], WORK, P.SPRITE, { cmd: "idle" });
  }
  assert.equal(play.phase, "rock");
  const beforeX = play.target.holdX;
  const moved = { ...WIN, x: WIN.x + 140 };
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { cmd: "idle" });
  assert.equal(play.phase, "rock");
  assert.ok(Math.abs(play.target.holdX - beforeX) > 40);
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { asleep: true, cmd: "sleep" });
  assert.ok(play.phase === "drop" || play.phase === "done" || play.phase === "rock-off");
  assert.equal(play.abort, true);
});
