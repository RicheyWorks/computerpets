
test("Pale sands a window stool as dry sand: walk onto the stool, run the pale, then leave", () => {
  const WIN_B = { id: "hw2", x: 980, y: 90, width: 300, height: 360 };
  assert.equal(P.playFor("ghost_crab"), "sand");
  assert.equal(P.SAND, "sand");
  assert.equal(P.playFor("fiddler_crab"), "signal");
  assert.equal(P.SIGNAL, "signal");
  assert.equal(P.playFor("amphipod"), "side");
  assert.equal(P.SIDE, "side");
  assert.equal(P.playFor("solifuge"), "run");
  assert.equal(P.RUN, "run");
  assert.equal(P.playFor("tardigrade"), "dry");
  assert.equal(P.DRY, "dry");
  assert.equal(P.playFor("hermit_crab"), "knob");
  assert.equal(P.playFor("horseshoe_crab"), "plow");
  assert.equal(P.playFor("saguaro"), "store");
  assert.equal(P.playFor("mining_bee"), "dig");
  assert.equal(P.playFor("squirrel"), "bury");
  assert.equal(P.playFor("vinegaroon"), "spray");
  assert.equal(P.playFor("limpet"), "sill");
  assert.notEqual(P.playFor("ghost_crab"), "pale");
  assert.notEqual(P.playFor("ghost_crab"), "run");
  assert.notEqual(P.playFor("ghost_crab"), "signal");
  assert.notEqual(P.playFor("ghost_crab"), "dry");
  assert.notEqual(P.playFor("ghost_crab"), "side");
  assert.notEqual(P.playFor("ghost_crab"), "plow");
  assert.notEqual(P.playFor("ghost_crab"), "store");
  assert.notEqual(P.playFor("ghost_crab"), "dig");
  assert.notEqual(P.playFor("ghost_crab"), "sill");
  assert.equal(P.DUR.signalOn, 2.45, "Wave signal durations stay");
  assert.equal(P.DUR.sideOn, 2.38, "Scud side durations stay");
  assert.equal(P.DUR.runOn, 0.44, "Gale run durations stay");
  assert.equal(P.DUR.dryOn, 2.18, "Tun dry durations stay");
  assert.equal(P.DUR.plowOn, 1.66, "Ledger plow durations stay");
  assert.equal(P.DUR.knobOn, 0.82, "Tenant knob durations stay");
  const target = P.pickTarget([WIN], 80, "ghost_crab", WORK, P.SPRITE);
  assert.ok(target);
  assert.equal(target.kind, "sand");
  assert.equal(target.side, "drysand");
  assert.equal(target.leave, "sands");
  assert.notEqual(target.kind, "pale");
  assert.notEqual(target.kind, "run");
  assert.notEqual(target.kind, "signal");
  assert.notEqual(target.kind, "plow");
  assert.notEqual(target.kind, "store");
  assert.notEqual(target.kind, "dig");
  assert.notEqual(target.kind, "sill");
  assert.ok(target.holdLift > 16, "she sands a window stool as dry sand, not the floor");
  assert.ok(P.DUR.sandHold > P.DUR.sand, "the hold is the sit after; the run on the sand is the tell");
  assert.ok(P.DUR.sandOn > 1.0, "a walk onto the stool, not the sand");
  assert.ok(P.DUR.sandOn !== P.DUR.signalOn);
  assert.ok(P.DUR.sandOn !== P.DUR.runOn);
  assert.ok(P.DUR.sandOn !== P.DUR.plowOn);
  assert.ok(P.DUR.sandOn !== P.DUR.storeOn);
  assert.ok(P.DUR.sandOn !== P.DUR.digOn);
  assert.ok(P.DUR.sandOn !== P.DUR.dryOn);
  assert.ok(P.DUR.sandOn !== P.DUR.sillHop);
  assert.ok(P.DUR.sand !== P.DUR.signal);
  assert.ok(P.DUR.sand !== P.DUR.run);
  assert.ok(P.DUR.sand !== P.DUR.plow);
  assert.ok(P.DUR.sandHold !== P.DUR.signalHold);
  assert.ok(P.DUR.sandHold !== P.DUR.runHold);
  assert.ok(P.DUR.sandOff !== P.DUR.signalOff);
  assert.ok(P.DUR.sandOff !== P.DUR.sillDown);
  const dish = P.sandPoint(WIN, P.SPRITE, WORK);
  const plow = P.plowPoint(WIN, P.SPRITE, WORK);
  const store = P.storePoint(WIN, P.SPRITE, WORK);
  const dig = P.digPoint(WIN, P.SPRITE, WORK);
  const bury = P.buryPoint(WIN, P.SPRITE, WORK);
  const spray = P.sprayPoint(WIN, P.SPRITE, WORK);
  const gale = P.runPoint(WIN, P.SPRITE, WORK);
  const wave = P.signalPoint(WIN, P.SPRITE, WORK);
  assert.ok(dish.lift > 16, "the window stool as dry sand, not the floor");
  assert.ok(Math.abs(dish.lift - plow.lift) < 2, "the same window stool Ledger plows; the pose is a sand");
  assert.ok(Math.abs(dish.x - plow.x) > 8, "not Ledger's window-stool plow");
  assert.ok(Math.abs(dish.x - store.x) > 8, "not Arm's window-stool store");
  assert.ok(Math.abs(dish.x - dig.x) > 8, "not Bank's window-stool dig");
  assert.ok(Math.abs(dish.x - bury.x) > 8, "not Cache's window-stool bury");
  assert.ok(Math.abs(dish.x - spray.x) > 8 || Math.abs(dish.lift - spray.lift) > 8, "not Whip's sill-wash spray");
  assert.ok(Math.abs(dish.x - gale.x) > 8 || Math.abs(dish.lift - gale.lift) > 8, "not Gale's sash-light run");
  assert.ok(Math.abs(dish.x - wave.x) > 8 || Math.abs(dish.lift - wave.lift) > 8, "not Wave's sill-pan signal");
  const tiny = P.pickTarget([{ id: "tiny", x: 200, y: 80, width: 120, height: 50 }], 80, "ghost_crab", WORK, P.SPRITE);
  assert.equal(tiny, null, "a real window stool, not a thin strip");
  const short = P.pickTarget([{ id: "short", x: 200, y: 80, width: 200, height: 80 }], 80, "ghost_crab", WORK, P.SPRITE);
  assert.equal(short, null, "a real window stool, not a short sash");
  const thin = P.pickTarget([{ id: "thin", x: 200, y: 80, width: 192, height: 156 }], 80, "ghost_crab", WORK, P.SPRITE);
  assert.equal(thin, null, "Pale needs Ledger's real window stool, not a shallower mouth");
  const stool = P.pickTarget([{ id: "stool", x: 200, y: 80, width: 192, height: 158 }], 80, "ghost_crab", WORK, P.SPRITE);
  assert.ok(stool, "a real window stool as dry sand");
  const ledgerOk = P.pickTarget([{ id: "stool", x: 200, y: 80, width: 192, height: 158 }], 80, "horseshoe_crab", WORK, P.SPRITE);
  assert.ok(ledgerOk, "Ledger still takes the stool");
  const armOk = P.pickTarget([{ id: "stool", x: 200, y: 80, width: 186, height: 176 }], 80, "saguaro", WORK, P.SPRITE);
  assert.ok(armOk, "Arm still takes the stool");
  const cacheOk = P.pickTarget([{ id: "stool", x: 200, y: 80, width: 194, height: 162 }], 80, "squirrel", WORK, P.SPRITE);
  assert.ok(cacheOk, "Cache still takes the stool");
  const galeOk = P.pickTarget([{ id: "light", x: 200, y: 80, width: 190, height: 174 }], 80, "solifuge", WORK, P.SPRITE);
  assert.ok(galeOk, "Gale still takes the sash light");
  const waveOk = P.pickTarget([{ id: "pan", x: 200, y: 80, width: 190, height: 184 }], 80, "fiddler_crab", WORK, P.SPRITE);
  assert.ok(waveOk, "Wave still takes the pan");
  const walkOn = P.sandOnPath(0.25, { x: 40, lift: 0 }, { x: dish.x, lift: dish.lift });
  const plowOn = P.plowOnPath(0.25, { x: 40, lift: 0 }, { x: dish.x, lift: dish.lift });
  const storeOn = P.storeOnPath(0.25, { x: 40, lift: 0 }, { x: dish.x, lift: dish.lift });
  const runOn = P.runOnPath(0.25, { x: 40, lift: 0 }, { x: dish.x, lift: dish.lift });
  const signalOn = P.signalOnPath(0.25, { x: 40, lift: 0 }, { x: dish.x, lift: dish.lift });
  assert.ok(walkOn.lift > 0, "she walks onto the stool as dry sand");
  assert.ok(walkOn.rot !== plowOn.rot, "a walk onto the stool, not Ledger plow");
  assert.ok(walkOn.rot !== storeOn.rot, "a walk onto the stool, not Arm store");
  assert.ok(walkOn.rot !== runOn.rot, "a walk onto the stool, not Gale run");
  assert.ok(walkOn.rot !== signalOn.rot, "a walk onto the stool, not Wave signal");
  const pulse = P.sandPath(0.5);
  const book = P.plowPath(0.5);
  const ribs = P.storePath(0.5);
  const shaft = P.digPath(0.5);
  const dash = P.runPath(0.5);
  const claw = P.signalPath(0.5);
  assert.ok(pulse.x > 6, "she runs the pale; the sand is the tell");
  assert.ok(pulse.lift > 0, "the pale runs the dry sand");
  assert.ok(pulse.rot !== book.rot, "a sand, not a plow");
  assert.ok(pulse.rot !== ribs.rot, "a sand, not a store");
  assert.ok(pulse.rot !== shaft.rot, "a sand, not a dig");
  assert.ok(pulse.rot !== dash.rot, "a sand, not a run");
  assert.ok(pulse.rot !== claw.rot, "a sand, not a signal");
  const hold = P.sandHoldPath(0.5);
  assert.ok(Math.abs(hold.rot - 9.2) < 0.2, "she holds after the run, pale still the tell");
  assert.ok(Math.abs(hold.x - 21.6) < 0.2, "she stays on the dry sand");
  assert.ok(hold.lift > 2, "a sit on the stool after the run");
  const off0 = P.sandOffPath(0, { x: dish.x, lift: dish.lift, rot: 9.2 }, { x: dish.x + 69, lift: 0 });
  const offMid = P.sandOffPath(0.5, { x: dish.x, lift: dish.lift, rot: 9.2 }, { x: dish.x + 69, lift: 0 });
  const off1 = P.sandOffPath(1, { x: dish.x, lift: dish.lift, rot: 9.2 }, { x: dish.x + 69, lift: 0 });
  assert.ok(Math.abs(off0.x - dish.x) < 2);
  assert.ok(Math.abs(offMid.x - dish.x) > 8, "a walk leave off the dry sand");
  assert.ok(Math.abs(off1.lift) < 3);
  let play = P.beginPlay(target, target.approachX);
  const seen = new Set();
  const windowIds = new Set();
  for (let i = 0; i < 2400 && play.phase !== "done"; i++) {
    seen.add(play.phase);
    if (play.target && play.target.id) windowIds.add(play.target.id);
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN, WIN_B], WORK, P.SPRITE, { cmd: "idle" });
    assert.notEqual(play.phase, "plow");
    assert.notEqual(play.phase, "store");
    assert.notEqual(play.phase, "dig");
    assert.notEqual(play.phase, "run");
    assert.notEqual(play.phase, "signal");
    assert.notEqual(play.phase, "sill-walk");
    if (play.phase === "sand") {
      assert.equal(play.anim, "play");
    }
    if (play.phase === "sand-on") {
      assert.equal(play.anim, "walk");
    }
    if (play.phase === "sand-hold" && play.t > 0) {
      assert.equal(play.anim, "sit");
      assert.ok(Math.abs(play.lift - (play.target.holdLift + 2.6)) < 3, "she holds the sit after the run on the stool");
    }
    if (play.phase === "sand-off") {
      assert.equal(play.anim, "walk");
    }
  }
  assert.ok(seen.has("sand-on"));
  assert.ok(seen.has("sand"));
  assert.ok(seen.has("sand-hold"));
  assert.ok(seen.has("sand-off"));
  assert.ok(!seen.has("plow"), "Pale never uses Ledger plow");
  assert.ok(!seen.has("store"), "Pale never uses Arm store");
  assert.ok(!seen.has("dig"), "Pale never uses Bank dig");
  assert.ok(!seen.has("run"), "Pale never uses Gale run");
  assert.ok(!seen.has("signal"), "Pale never uses Wave signal");
  assert.equal(windowIds.size, 1, "one window");
  assert.equal(play.phase, "done");
  assert.equal(play.lift, 0);
});
test("a moved window refits Pale's dry-sand sand; sleep, card, and hide abort; Pale never starts asleep", () => {
  assert.equal(P.canStart({ asleep: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ card: true, cmd: "idle" }), false);
  assert.equal(P.canStart({ hidden: true, cmd: "wander" }), false);
  assert.equal(P.shouldAbort({ cmd: "sleep" }), true);
  assert.equal(P.shouldAbort({ cmd: "hide" }), true);
  assert.equal(P.shouldAbort({ card: true, cmd: "idle" }), true);
  const target = P.pickTarget([WIN], 200, "ghost_crab", WORK, P.SPRITE);
  let play = P.beginPlay(target, target.approachX);
  for (let i = 0; i < 900 && play.phase !== "sand"; i++) {
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN], WORK, P.SPRITE, { cmd: "idle" });
  }
  assert.equal(play.phase, "sand");
  const beforeX = play.target.holdX;
  const moved = { ...WIN, x: WIN.x + 140 };
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { cmd: "idle" });
  assert.equal(play.phase, "sand");
  assert.ok(Math.abs(play.target.holdX - beforeX) > 40);
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { asleep: true, cmd: "sleep" });
  assert.ok(play.phase === "drop" || play.phase === "done" || play.phase === "sand-off");
  assert.equal(play.abort, true);
});
