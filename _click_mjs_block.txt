test("Click rights a window stool as a bark plate: walk onto the stool, sit, right once, sit the plate, then leave", () => {
  const WIN_B = { id: "hw2", x: 980, y: 90, width: 300, height: 360 };
  assert.equal(P.playFor("click_beetle"), "right");
  assert.equal(P.RIGHT, "right");
  assert.notEqual(P.playFor("click_beetle"), "click");
  assert.notEqual(P.playFor("click_beetle"), "snap");
  assert.notEqual(P.playFor("click_beetle"), "flip");
  assert.notEqual(P.playFor("click_beetle"), "glow");
  assert.notEqual(P.playFor("click_beetle"), "drill");
  assert.notEqual(P.playFor("click_beetle"), "sill");
  assert.equal(P.playFor("acorn_weevil"), "drill");
  assert.equal(P.DRILL, "drill");
  assert.equal(P.playFor("relay_dragon"), "click");
  assert.equal(P.CLICK, "click");
  assert.equal(P.playFor("venus_flytrap"), "count");
  assert.equal(P.playFor("snapper"), "snap");
  assert.equal(P.playFor("hognose"), "flip");
  assert.equal(P.playFor("firefly"), "glow");
  assert.equal(P.playFor("robber_fly"), "sill");
  assert.equal(Overlay.playFor("click_beetle"), "right");
  const target = P.pickTarget([WIN], 80, "click_beetle", WORK, P.SPRITE);
  assert.ok(target);
  assert.equal(target.kind, "right");
  assert.equal(target.side, "barkplate");
  assert.equal(target.leave, "righted");
  assert.ok(target.holdLift >= 0);
  assert.ok(P.DUR.rightHold > P.DUR.right * 1.2);
  assert.ok(P.DUR.rightOn !== P.DUR.drillOn);
  assert.ok(P.DUR.rightOn !== P.DUR.flipOn);
  assert.ok(P.DUR.right !== P.DUR.drill);
  assert.ok(P.DUR.right !== P.DUR.flip);
  const plate = P.rightPoint(WIN, P.SPRITE, WORK);
  const cup = P.drillPoint(WIN, P.SPRITE, WORK);
  const bark = P.cerciPoint(WIN, P.SPRITE, WORK);
  const flip = P.flipPoint(WIN, P.SPRITE, WORK);
  assert.ok(Math.abs(plate.x - cup.x) > 12 || Math.abs(plate.lift - cup.lift) > 2);
  assert.ok(Math.abs(plate.x - bark.x) > 12 || Math.abs(plate.lift - bark.lift) > 2);
  assert.ok(Math.abs(plate.x - flip.x) > 12 || Math.abs(plate.lift - flip.lift) > 2);
  const tiny = P.pickTarget([{ id: "tiny", x: 200, y: 80, width: 120, height: 50 }], 80, "click_beetle", WORK, P.SPRITE);
  assert.equal(tiny, null);
  const okRight = P.pickTarget([{ id: "right", x: 200, y: 80, width: 184, height: 152 }], 80, "click_beetle", WORK, P.SPRITE);
  assert.ok(okRight);
  const rightPose = P.rightPath(0.3);
  assert.ok(rightPose.lift > 4 || Math.abs(rightPose.rot) > 40);
  const hold = P.rightHoldPath(0.5);
  assert.ok(Math.abs(hold.rot + 2.8) < 0.2);
  let play = P.beginPlay(target, target.approachX);
  const seen = new Set();
  for (let i = 0; i < 2400 && play.phase !== "done"; i++) {
    seen.add(play.phase);
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN, WIN_B], WORK, P.SPRITE, { cmd: "idle" });
    assert.notEqual(play.phase, "drill");
    assert.notEqual(play.phase, "flip");
    assert.notEqual(play.phase, "jump");
    assert.notEqual(play.phase, "snap");
    assert.notEqual(play.phase, "glow");
  }
  assert.ok(seen.has("right-on"));
  assert.ok(seen.has("right"));
  assert.ok(seen.has("right-hold"));
  assert.ok(seen.has("right-off"));
  assert.equal(play.phase, "done");
});

test("a moved window refits Click's bark-plate right; sleep, card, and hide abort", () => {
  assert.equal(P.canStart({ asleep: true, cmd: "idle" }), false);
  const target = P.pickTarget([WIN], 200, "click_beetle", WORK, P.SPRITE);
  let play = P.beginPlay(target, target.approachX);
  for (let i = 0; i < 900 && play.phase !== "right"; i++) {
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN], WORK, P.SPRITE, { cmd: "idle" });
  }
  assert.equal(play.phase, "right");
  const beforeX = play.target.holdX;
  const moved = { ...WIN, x: WIN.x + 140 };
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { cmd: "idle" });
  assert.equal(play.phase, "right");
  assert.ok(Math.abs(play.target.holdX - beforeX) > 40);
  play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [moved], WORK, P.SPRITE, { asleep: true, cmd: "sleep" });
  assert.ok(play.phase === "drop" || play.phase === "done" || play.phase === "right-off");
  assert.equal(play.abort, true);
});
