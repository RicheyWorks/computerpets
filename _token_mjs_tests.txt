test("the demo window plate walks Token flat the same way", () => {
  const WIN_B = { id: "hw2", x: 980, y: 90, width: 300, height: 360 };
  assert.match(demoSrc, /demoWindow/);
  assert.match(livingSrc, /playFor/);
  assert.match(readFileSync(join(root, "src/lib/pets/shore.ts"), "utf8"), /key: "sand_dollar"[\s\S]{0,80}slug: "token"/);
  assert.equal(P.playFor("sand_dollar"), "flat");
  const target = P.pickTarget([WIN], 80, "sand_dollar", WORK, 176);
  assert.ok(target);
  assert.equal(target.kind, "flat");
  assert.equal(target.side, "sandplate");
  assert.equal(target.leave, "flats");
  assert.equal(Overlay.playFor("sand_dollar"), "flat");
  assert.equal(P.FLAT, "flat");
  assert.equal(Overlay.FLAT, "flat");
  assert.notEqual(P.playFor("sand_dollar"), "token");
  assert.notEqual(P.playFor("sand_dollar"), "bury");
  assert.notEqual(P.playFor("sand_dollar"), "sand");
  assert.notEqual(P.playFor("sand_dollar"), "circle");
  assert.notEqual(P.playFor("sand_dollar"), "open");
  assert.notEqual(P.playFor("sand_dollar"), "reef");
  assert.equal(P.playFor("squirrel"), "bury");
  assert.equal(Overlay.playFor("squirrel"), "bury");
  assert.equal(P.playFor("ghost_crab"), "sand");
  assert.equal(Overlay.playFor("ghost_crab"), "sand");
  assert.equal(P.playFor("periwinkle"), "rock");
  assert.equal(Overlay.playFor("periwinkle"), "rock");
  assert.equal(P.playFor("chiton"), "eight");
  assert.equal(Overlay.playFor("chiton"), "eight");
  assert.equal(P.playFor("barnacle"), "cirri");
  assert.equal(Overlay.playFor("barnacle"), "cirri");
  assert.equal(P.playFor("limpet"), "clamp");
  assert.equal(Overlay.playFor("limpet"), "clamp");
  assert.equal(P.playFor("fiddler_crab"), "signal");
  assert.equal(Overlay.playFor("fiddler_crab"), "signal");
  assert.equal(P.playFor("goldfish"), "circle");
  assert.equal(Overlay.playFor("goldfish"), "circle");
  assert.equal(P.playFor("water_lily"), "open");
  assert.equal(Overlay.playFor("water_lily"), "open");
  assert.equal(P.playFor("sea_star"), "reef");
  assert.equal(Overlay.playFor("sea_star"), "reef");
  assert.equal(P.playFor("sea_urchin"), "sill");
  assert.equal(P.DUR.flatOn, Overlay.DUR.flatOn);
  assert.equal(P.DUR.flat, Overlay.DUR.flat);
  assert.equal(P.DUR.flatHold, Overlay.DUR.flatHold);
  assert.equal(P.DUR.flatOff, Overlay.DUR.flatOff);
  assert.ok(P.DUR.flatOn !== Overlay.DUR.rockOn);
  assert.ok(P.DUR.flatOn !== Overlay.DUR.signalOn);
  assert.ok(P.DUR.flatOn !== Overlay.DUR.sandOn);
  assert.ok(P.DUR.flatOn !== Overlay.DUR.buryOn);
  const plate = P.flatPoint(WIN, 176, WORK);
  const deskPlate = Overlay.flatPoint(WIN, Overlay.SPRITE, WORK);
  const marsh = P.signalPoint(WIN, 176, WORK);
  const current = P.paddlePoint(WIN, 176, WORK);
  const bowl = P.rinsePoint(WIN, 176, WORK);
  const dry = P.sandPoint(WIN, 176, WORK);
  const oak = P.buryPoint(WIN, 176, WORK);
  assert.ok(Math.abs(plate.x - deskPlate.x) < 1);
  assert.ok(Math.abs(plate.lift - deskPlate.lift) < 1);
  assert.ok(Math.abs(plate.lift - marsh.lift) < 2, "same sill pan as Wave, different pose");
  assert.ok(Math.abs(plate.x - marsh.x) > 8, "not Wave signal");
  assert.ok(Math.abs(plate.lift - current.lift) < 2, "same sill pan as Spoon, different pose");
  assert.ok(Math.abs(plate.x - current.x) > 8, "not Spoon paddle");
  assert.ok(plate.lift > 28, "sand plate, the pan");
  assert.ok(Math.abs(plate.x - dry.x) > 8 || Math.abs(plate.lift - dry.lift) > 8, "not Pale sand");
  assert.ok(Math.abs(plate.x - oak.x) > 8 || Math.abs(plate.lift - oak.lift) > 8, "not Cache bury");
  const tiny = P.pickTarget([{ id: "tiny", x: 200, y: 80, width: 120, height: 50 }], 80, "sand_dollar", WORK, 176);
  assert.equal(tiny, null, "a real sill pan");
  const okPan = P.pickTarget([{ id: "pan", x: 200, y: 80, width: 190, height: 184 }], 80, "sand_dollar", WORK, 176);
  assert.ok(okPan, "a real sill pan as a sand plate");
  const walkOn = P.flatOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  const deskWalk = Overlay.flatOnPath(0.25, { x: 40, lift: 0 }, { x: plate.x, lift: plate.lift });
  assert.equal(walkOn.x, deskWalk.x);
  assert.equal(walkOn.lift, deskWalk.lift);
  const pulse = P.flatPath(0.5);
  const deskPulse = Overlay.flatPath(0.5);
  assert.equal(pulse.x, deskPulse.x);
  assert.ok(pulse.lift < 0, "a flat sit, the play");
  assert.ok(pulse.rot !== P.signalPath(0.5).rot, "a flat, not a signal");
  assert.ok(pulse.rot !== P.sandPath(0.5).rot, "a flat, not a sand");
  assert.ok(pulse.rot !== P.buryPath(0.5).rot, "a flat, not a bury");
  assert.ok(pulse.rot !== P.rockPath(0.5).rot, "a flat, not a rock");
  const hold = P.flatHoldPath(0.5);
  const deskHold = Overlay.flatHoldPath(0.5);
  assert.equal(hold.x, deskHold.x);
  assert.equal(hold.lift, deskHold.lift);
  let play = P.beginPlay(target, target.approachX);
  const seen = new Set();
  const windowIds = new Set();
  for (let i = 0; i < 2400 && play.phase !== "done"; i++) {
    seen.add(play.phase);
    if (play.target && play.target.id) windowIds.add(play.target.id);
    play = P.stepPlay(play, 0.05, { x: play.x, lift: play.lift }, [WIN, WIN_B], WORK, 176, { cmd: "idle" });
    if (play.phase === "flat") {
      assert.equal(play.anim, "play");
    }
    if (play.phase === "flat-on") {
      assert.equal(play.anim, "walk");
    }
    if (play.phase === "flat-hold" && play.t > 0) {
      assert.equal(play.anim, "sit");
      assert.ok(Math.abs(play.lift - (play.target.holdLift - 3.2)) < 3, "she holds the flat after the sit on the pan");
    }
  }
  assert.ok(seen.has("flat-on"));
  assert.ok(seen.has("flat"));
  assert.ok(seen.has("flat-hold"));
  assert.ok(seen.has("flat-off"));
  assert.equal(windowIds.size, 1, "one window");
  assert.equal(play.phase, "done");
});
